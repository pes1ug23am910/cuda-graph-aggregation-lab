cmake_minimum_required(VERSION 3.25)
project(graph_aggregation_lab VERSION 1.0.0 LANGUAGES CXX)
option(GRAPHLAB_CUDA "Build CUDA kernels and cuSPARSE backend" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(graphlab_core src/core.cpp)
target_include_directories(graphlab_core PUBLIC include)
if(MSVC)
  target_compile_options(graphlab_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(graphlab_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
add_executable(graphlab src/main.cpp)
target_link_libraries(graphlab PRIVATE graphlab_core)
if(GRAPHLAB_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(graphlab_cuda src/cuda.cu)
  target_include_directories(graphlab_cuda PUBLIC include)
  target_link_libraries(graphlab_cuda PUBLIC graphlab_core CUDA::cudart CUDA::cusparse)
  set_target_properties(graphlab_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(graphlab_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ftz=false;--prec-div=true;--prec-sqrt=true;--fmad=false;-lineinfo>)
  target_link_libraries(graphlab PRIVATE graphlab_cuda)
  target_compile_definitions(graphlab PRIVATE GRAPHLAB_CUDA=1)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(graphlab_core_tests tests/core_tests.cpp)
  target_link_libraries(graphlab_core_tests PRIVATE graphlab_core)
  add_test(NAME core_contract COMMAND graphlab_core_tests)
  add_test(NAME cli_verify_cpu COMMAND graphlab verify --backend cpu)
  add_test(NAME cli_fixture_roundtrip COMMAND graphlab roundtrip)
  if(GRAPHLAB_CUDA)
    add_test(NAME cuda_contract COMMAND graphlab verify --backend all)
    set_tests_properties(cuda_contract PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
